// ==============================
// 1. Format Number
// ==============================

const number = 1299999;

const formattedNumber = number.toLocaleString("en-US");

console.log(formattedNumber);
// 1,299,999


// ==============================
// 2. Format Currency - USD
// ==============================

const price = 1299;

const formattedPrice = price.toLocaleString("en-US", {
  style: "currency",
  currency: "USD",
});

console.log(formattedPrice);
// $1,299.00


// ==============================
// 3. Format Currency - EGP
// ==============================

const egpPrice = 1500;

const formattedEGP = egpPrice.toLocaleString("en-EG", {
  style: "currency",
  currency: "EGP",
});

console.log(formattedEGP);
// EGP 1,500.00


// ==============================
// 4. Format Date
// ==============================

const date = new Date("2026-08-31");

const formattedDate = date.toLocaleDateString("en-US");

console.log(formattedDate);
// 8/31/2026


// ==============================
// 5. Format Date - Friendly
// ==============================

const friendlyDate = date.toLocaleDateString("en-US", {
  year: "numeric",
  month: "short",
  day: "numeric",
});

console.log(friendlyDate);
// Aug 31, 2026


// ==============================
// 6. Format Date - Full
// ==============================

const fullDate = date.toLocaleDateString("en-US", {
  year: "numeric",
  month: "long",
  day: "numeric",
});

console.log(fullDate);
// August 31, 2026


// ==============================
// 7. Format Time
// ==============================

const time = new Date("2026-08-31T17:30:00");

const formattedTime = time.toLocaleTimeString("en-US");

console.log(formattedTime);
// 5:30:00 PM


// ==============================
// 8. Format Time - Without Seconds
// ==============================

const simpleTime = time.toLocaleTimeString("en-US", {
  hour: "2-digit",
  minute: "2-digit",
});

console.log(simpleTime);
// 05:30 PM


// ==============================
// 9. Format Date + Time
// ==============================

const dateTime = time.toLocaleString("en-US", {
  year: "numeric",
  month: "short",
  day: "numeric",
  hour: "2-digit",
  minute: "2-digit",
});

console.log(dateTime);
// Aug 31, 2026, 05:30 PM


// ==============================
// 10. Convert KM → Miles
// ==============================

const kilometers = 100;

const miles = kilometers * 0.621371;

console.log(miles);
// 62.1371


// ==============================
// 11. Convert Miles → KM
// ==============================

const milesValue = 100;

const km = milesValue * 1.60934;

console.log(km);
// 160.934


// ==============================
// 12. Convert Meters → KM
// ==============================

const meters = 5000;

const kilometersValue = meters / 1000;

console.log(kilometersValue);
// 5


// ==============================
// 13. Convert KM → Meters
// ==============================

const kmValue = 5;

const metersValue = kmValue * 1000;

console.log(metersValue);
// 5000


// ==============================
// 14. Convert Celsius → Fahrenheit
// ==============================

const celsius = 25;

const fahrenheit = (celsius * 9) / 5 + 32;

console.log(fahrenheit);
// 77


// ==============================
// 15. Convert Fahrenheit → Celsius
// ==============================

const fahrenheitValue = 77;

const celsiusValue = ((fahrenheitValue - 32) * 5) / 9;

console.log(celsiusValue);
// 25

// ==============================
// 22. Convert Minutes → Hours
// ==============================

const minutes = 150;

const hours = Math.floor(minutes / 60);
const remainingMinutes = minutes % 60;

console.log(`${hours}h ${remainingMinutes}m`);
// 2h 30m


// ==============================
// 23. Convert Seconds → Minutes
// ==============================

const seconds = 150;

const mins = Math.floor(seconds / 60);
const remainingSeconds = seconds % 60;

console.log(`${mins}m ${remainingSeconds}s`);
// 2m 30s
